public static boolean check(int[] nums) {
    int n = nums.length;
    
    int j = -1;
    
    
    
    for( int i = 1; i < n; i++){
        if(nums[i] < nums[i - 1]){
            j = i;
            break;
        }
    }
    
    if ( j == -1) return true;
    
    for( int i = j; i < n - 1; i++){
        
        if( nums[i] > nums[i+1]){
            return false;
        }
    } 
           
    if (nums[n - 1] > nums[0]){
        return false;
    }
    
    return true;
        
}
