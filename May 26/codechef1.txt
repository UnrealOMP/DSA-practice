    public static int findMaxConsecutiveOnes(int[] nums) {
       int i = 0;
       int maxLen = 0;
       
       while(i < nums.length){
           int count = 0;
          if(nums[i] == 1){
              int j = i;
              while(j < nums.length && nums[j] == 1){
                  count++;
                  j++;
              }
              
          }
          maxLen = Math.max(maxLen, count);
          i++;
       }
       return maxLen;
       
    }
