import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        int T = sc.nextInt();
        
        while(T > 0){
            int N = sc.nextInt();
            int nums[] = new int[N];
            
            for(int i = 0; i < N; i++){
                nums[i] = sc.nextInt();
            }
            int count = N;
            for(int j = 1; j < N; j++){
                if(N == 1){
                    return;
                }
                if(nums[j] == nums[j-1]){
                    count--;
                }
            }
            System.out.println(count);
            T--;
            
        }
        
    }
}
